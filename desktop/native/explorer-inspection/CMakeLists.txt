cmake_minimum_required(VERSION 3.24)
project(FurtherComparePackageInspection LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC)
 message(FATAL_ERROR "Requires Visual Studio C++ and current Windows SDK; no cross-platform attestation claims.")
endif()
add_executable(FurtherComparePackageInspection PackageInspection.cpp)
target_compile_features(FurtherComparePackageInspection PRIVATE cxx_std_17)
target_compile_definitions(FurtherComparePackageInspection PRIVATE UNICODE _UNICODE WINVER=0x0A00 _WIN32_WINNT=0x0A00)
target_compile_options(FurtherComparePackageInspection PRIVATE /W4 /permissive- /EHsc /guard:cf /utf-8)
target_link_options(FurtherComparePackageInspection PRIVATE /DYNAMICBASE /NXCOMPAT /guard:cf)
target_link_libraries(FurtherComparePackageInspection PRIVATE ole32 shlwapi xmllite wintrust crypt32 advapi32 kernel32)
