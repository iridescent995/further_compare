cmake_minimum_required(VERSION 3.24)
project(FurtherCompareExplorer VERSION 1.0.0 LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC)
  message(FATAL_ERROR "Build with Visual Studio 2022 C++ and the Windows SDK on Windows. This is not a cross-compile/runtime validation target.")
endif()
add_library(FurtherCompareExplorer SHARED ExplorerCommand.cpp ExplorerCommand.def)
# Optional deterministic assets are generated on macOS/Windows before SDK compilation.
# This embeds resource 101 only; GetIcon activation remains a separate native change.
set(FC_EXPLORER_ASSETS_DIR "" CACHE PATH "Directory produced by build-explorer-assets.cjs")
if(FC_EXPLORER_ASSETS_DIR)
 get_filename_component(FC_EXPLORER_ASSETS_DIR "${FC_EXPLORER_ASSETS_DIR}" ABSOLUTE)
 if(FC_EXPLORER_ASSETS_DIR MATCHES "[\"\r\n]")
  message(FATAL_ERROR "Explorer asset path contains unsupported resource-script characters.")
 endif()
 foreach(asset FurtherCompareExplorer.ico Assets/StoreLogo.png Assets/Square44x44Logo.png Assets/Square150x150Logo.png)
  if(NOT EXISTS "${FC_EXPLORER_ASSETS_DIR}/${asset}" OR IS_DIRECTORY "${FC_EXPLORER_ASSETS_DIR}/${asset}")
   message(FATAL_ERROR "Missing generated Explorer asset: ${asset}")
  endif()
 endforeach()
 enable_language(RC)
 set(FC_EXPLORER_ICON "${FC_EXPLORER_ASSETS_DIR}/FurtherCompareExplorer.ico")
 configure_file(ExplorerIcon.rc.in "${CMAKE_CURRENT_BINARY_DIR}/ExplorerIcon.rc" @ONLY)
 target_sources(FurtherCompareExplorer PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/ExplorerIcon.rc")
 file(COPY "${FC_EXPLORER_ASSETS_DIR}/Assets" DESTINATION "${CMAKE_CURRENT_BINARY_DIR}")
endif()
target_compile_features(FurtherCompareExplorer PRIVATE cxx_std_17)
target_compile_definitions(FurtherCompareExplorer PRIVATE UNICODE _UNICODE WINVER=0x0A00 _WIN32_WINNT=0x0A00)
target_compile_options(FurtherCompareExplorer PRIVATE "$<$<COMPILE_LANGUAGE:CXX>:/W4;/permissive-;/EHsc;/guard:cf;/utf-8>")
target_link_options(FurtherCompareExplorer PRIVATE /DYNAMICBASE /NXCOMPAT /guard:cf)
target_link_libraries(FurtherCompareExplorer PRIVATE ole32 shlwapi advapi32 uuid)
set_property(TARGET FurtherCompareExplorer PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
# Configure identity only when the deploying owner provides it; never ship a
# fabricated publisher identity or auto-register an unsigned package.
set(FC_PACKAGE_NAME "" CACHE STRING "Assigned sparse package identity name")
set(FC_PACKAGE_PUBLISHER "" CACHE STRING "Certificate-matching publisher distinguished name")
set(FC_PACKAGE_VERSION "1.0.0.0" CACHE STRING "Four-component package version")
if(CMAKE_GENERATOR_PLATFORM STREQUAL "ARM64")
 set(FC_PACKAGE_ARCH "arm64")
elseif(CMAKE_GENERATOR_PLATFORM STREQUAL "x64")
 set(FC_PACKAGE_ARCH "x64")
else()
 message(FATAL_ERROR "Select -A x64 or -A ARM64, matching target Explorer architecture.")
endif()
if(NOT FC_PACKAGE_VERSION MATCHES "^[0-9]+\\.[0-9]+\\.[0-9]+\\.[0-9]+$")
 message(FATAL_ERROR "Package version must contain four numeric components.")
endif()
if(FC_PACKAGE_NAME AND FC_PACKAGE_PUBLISHER)
 if(FC_PACKAGE_NAME MATCHES "[<>\"&]" OR FC_PACKAGE_PUBLISHER MATCHES "[<>\"&]")
  message(FATAL_ERROR "Identity contains XML metacharacters; provide an escaped/validated deployment identity through a dedicated packaging step.")
 endif()
 configure_file(Package.appxmanifest.in ${CMAKE_CURRENT_BINARY_DIR}/Package.appxmanifest @ONLY)
endif()
include(CTest)
if(BUILD_TESTING)
 add_executable(ExplorerComContractTests ComContractTests.cpp)
 target_compile_features(ExplorerComContractTests PRIVATE cxx_std_17)
 target_compile_definitions(ExplorerComContractTests PRIVATE UNICODE _UNICODE)
 # Assertions are the test oracle, including Release builds.
 target_compile_options(ExplorerComContractTests PRIVATE /W4 /UNDEBUG /utf-8)
 target_link_libraries(ExplorerComContractTests PRIVATE ole32 uuid)
 add_test(NAME ExplorerComContract COMMAND ExplorerComContractTests $<TARGET_FILE:FurtherCompareExplorer>)
endif()
